/* ===================================== V19 SPEECH RECOGNITION ===================================== */ function setupRecognition() { const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (!SpeechRecognition) { alert( "Speech recognition is not supported by this browser. Please use Google Chrome or Microsoft Edge." ); return null; } const r = new SpeechRecognition(); r.continuous = true; /* IMPORTANT: Keep interim results ON. iPhone may change/revise interim transcripts before they become final. We therefore track each recognition result separately. */ r.interimResults = true; r.maxAlternatives = 1; r.lang = recognitionLanguage; /* ===================================== SPEECH TRACKING ===================================== */ let resultCounts = {}; let finalResults = {}; /* ===================================== RECOGNITION START ===================================== */ r.onstart = () => { recognitionRunning = true; recognitionStarting = false; if (!listening) { try { r.stop(); } catch(e) {} return; } setListeningStatus(); }; /* ===================================== RECOGNITION RESULT ===================================== */ r.onresult = event => { if (!listening) { return; } for ( let i = event.resultIndex; i < event.results.length; i++ ) { const result = event.results[i]; if ( !result || !result[0] ) { continue; } const transcript = result[0].transcript.trim(); if (!transcript) { continue; } /* Show what iPhone heard */ heardText.textContent = "Heard: " + transcript; console.log( "Recognized:", transcript, "Final:", result.isFinal ); /* ===================================== COUNT CURRENT RESULT ===================================== */ const currentCount = countWaheguruMatches( transcript ); const previousCount = resultCounts[i] || 0; /* Only add the NEW matches. Example: First: "waheguru" = 1 Then iPhone changes it to: "waheguru waheguru" = 2 We add only: 2 - 1 = 1 Then: "waheguru waheguru waheguru" = 3 We add only: 3 - 2 = 1 */ if ( currentCount > previousCount ) { const newCount = currentCount - previousCount; addCount( newCount ); } /* Remember the largest count we have already processed. */ if ( currentCount > previousCount ) { resultCounts[i] = currentCount; } /* ===================================== FINAL RESULT ===================================== */ if (result.isFinal) { finalResults[i] = true; /* Keep the final count locked. */ resultCounts[i] = currentCount; } } }; /* ===================================== RECOGNITION ERROR ===================================== */ r.onerror = event => { console.log( "Speech recognition:", event.error ); if ( event.error === "not-allowed" || event.error === "service-not-allowed" ) { listening = false; clearTimers(); stopAudioMonitor(); setPausedStatus( "Microphone permission needed" ); return; } if ( event.error === "aborted" ) { return; } if (listening) { scheduleRecognitionRestart(); } }; /* ===================================== RECOGNITION END ===================================== */ r.onend = () => { recognitionRunning = false; recognitionStarting = false; if (!listening) { return; } r.lang = recognitionLanguage; /* New recognition session. Clear the old result indexes because iPhone may reuse result indexes after recognition restarts. */ resultCounts = {}; finalResults = {}; scheduleRecognitionRestart(); }; return r; }
